cmake_minimum_required(VERSION 3.16)
project(FlintControllerChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
find_package(OpenSSL REQUIRED)
enable_testing()
add_executable(controller-checks controller_test.cpp ../../flint/flintController.cpp ../../flint/flintController.h ../../flint/flintQrImage.cpp ../../flint/flintHealth.cpp ../../flint/flintUpdates.cpp ../../flint/flintUpdates.h)
target_include_directories(controller-checks PRIVATE . ../../flint)
target_link_libraries(controller-checks PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test OpenSSL::Crypto)
qt_add_resources(controller-checks flint_routing PREFIX "/ui/qml/Assets" BASE "../../flint" FILES "../../flint/flint-routing-catalog.json")
add_test(NAME controller-checks COMMAND controller-checks)
add_executable(update-checks update_test.cpp ../../flint/flintController.cpp ../../flint/flintController.h ../../flint/flintQrImage.cpp ../../flint/flintHealth.cpp ../../flint/flintUpdates.cpp ../../flint/flintUpdates.h)
target_include_directories(update-checks PRIVATE . ../../flint)
target_link_libraries(update-checks PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test OpenSSL::Crypto)
qt_add_resources(update-checks update_routing PREFIX "/ui/qml/Assets" BASE "../../flint" FILES "../../flint/flint-routing-catalog.json")
add_test(NAME update-checks COMMAND update-checks)
add_executable(telemetry-checks telemetry_test.cpp)
target_include_directories(telemetry-checks PRIVATE ../../flint)
target_link_libraries(telemetry-checks PRIVATE Qt6::Core Qt6::Network Qt6::Test)
add_test(NAME telemetry-checks COMMAND telemetry-checks)

add_executable(balance-checks balance_test.cpp)
